const owner=journalOwnerSql(query=>execFileSync('docker',['exec','-i',target.identity.databaseContainerId,'psql','-X','-q','-A','-t','-v','ON_ERROR_STOP=1','-U','post'+'gres','-d','post'+'gres'],{input:query,encoding:'utf8',timeout:30000}).trim(),transport,privatePath);
const sql=query=>{assert(/^begin read only;/i.test(query.trim()));assert(/commit;\s*$/i.test(query));return JSON.parse(owner(query));};
const observations=[];
for(const phase of m.fixture.phases){
 const ref=r.references.phases[phase.phase],lineage={...r.origin,phase:phase.phase,namespace:phase.plannedNamespace};
 for(const j of ref.journals)assert.equal(hash(readFileSync(j.path)),j.sha256);
 const original=new FixtureJournal(resolve(ref.privateDirectory,'owned-journal.jsonl'),lineage,m.fixture.rowCaps[phase.phase]);original.assertReady();
 const old=bound(ref.inventory),inventory=sql(fixtureInventorySql(m.fixture.rowScopes,original.owned()));assert.deepEqual(inventory,old.inventory);assertInventoryCaps(inventory,m.fixture.rowCaps[phase.phase]);
 observations.push({phase:phase.phase,families:Object.keys(inventory).length,allIdsHashesEqual:true,allCapsPass:true,authSubjects:inventory.auth_users.count,inventory});
}
const schema=bound(r.references.schema),norm=v=>Array.isArray(v)?v.map(norm).sort((a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b))):v&&typeof v==='object'?Object.fromEntries(Object.keys(v).sort().map(k=>[k,norm(v[k])])):v;

transport.assertReady();assert(!existsSync(namespace));
