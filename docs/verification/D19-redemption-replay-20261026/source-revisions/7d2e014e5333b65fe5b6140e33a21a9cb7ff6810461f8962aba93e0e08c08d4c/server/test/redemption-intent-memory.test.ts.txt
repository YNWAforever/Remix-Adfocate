async function snapshot():Promise<Snapshot>{
  const r=await db.query<{state:Snapshot}>(`select jsonb_build_object(
    'balances',(select jsonb_agg(jsonb_build_object('creator_id',creator_id,'balance',balance)order by creator_id)from public.point_balances),
    'ledger',(select coalesce(jsonb_agg(jsonb_build_object('creator_id',creator_id,'delta',delta)order by created_at,id),'[]'::jsonb)from public.point_transactions),
    'stock',(select jsonb_agg(jsonb_build_object('id',id,'stock_remaining',stock_remaining)order by id)from public.redemption_gifts),
    'codes',(select jsonb_agg(jsonb_build_object('code',code,'status',status,'issued_to',issued_to)order by code)from public.redemption_codes),
    'redemptions',(select coalesce(jsonb_agg(to_jsonb(r)order by created_at,id),'[]'::jsonb)from public.redemptions r))as state`);
  return r.rows[0].state;
}

describe("UC-C33 current actual redemption SQL/route intent replay",()=>{
  it("same intent after a lost response returns the same saved code and debits only once",async()=>{
    const first=await post(R1);expect(first.status).toBe(200);const retry=await post(R1);const state=await snapshot();events.push({readback:state});
    expect(retry.status).toBe(200);expect(retry.result.id).toBe(first.result.id);expect(retry.result.code).toBe(first.result.code);
    expect(state.balances.find(r=>r.creator_id===A)!.balance).toBe(150);expect(state.ledger).toHaveLength(1);expect(state.redemptions).toHaveLength(1);expect(state.stock.find(r=>r.id===G)!.stock_remaining).toBe(3);
  });
  it("a deliberate second new intent still receives a DISTINCT code and its own debit",async()=>{
    const first=await post(R1);const second=await post(R2);expect(first.status).toBe(200);expect(second.status).toBe(200);expect(second.result.id).not.toBe(first.result.id);expect(second.result.code).not.toBe(first.result.code);
    const state=await snapshot();expect(state.balances.find(r=>r.creator_id===A)!.balance).toBe(100);expect(state.ledger).toHaveLength(2);expect(state.redemptions).toHaveLength(2);
  });
  it("retry recovers the original saved result even when that first action exhausted stock",async()=>{
    await db.query("update public.redemption_gifts set stock_remaining=1 where id=$1",[G]);const first=await post(R1);expect(first.status).toBe(200);const retry=await post(R1);events.push({readback:await snapshot()});
    expect(retry.status).toBe(200);expect(retry.result.id).toBe(first.result.id);expect((await snapshot()).ledger).toHaveLength(1);
  });
  it("the same intent cannot be changed to another gift and debit a second time",async()=>{
    expect((await post(R1)).status).toBe(200);const changed=await post(R1,G2);events.push({readback:await snapshot()});expect(changed.status).toBe(409);expect(changed.result.code).toBe("request_conflict");expect((await snapshot()).ledger).toHaveLength(1);
  });
  it("a failed debit rolls back stock/code and the same intent can succeed after the balance fixture changes",async()=>{
    await db.query("update public.point_balances set balance=0 where creator_id=$1",[A]);const before=await snapshot();expect((await post(R1)).status).toBe(409);expect(await snapshot()).toEqual(before);
    await db.query("update public.point_balances set balance=100 where creator_id=$1",[A]);expect((await post(R1)).status).toBe(200);expect((await snapshot()).ledger).toHaveLength(1);
  });
  it("another recipient using the same client UUID never receives the first person's code",async()=>{
    const first=await post(R1);const other=await post(R1,G,B);expect(first.status).toBe(200);expect(other.status).toBe(200);expect(other.result.creator_id).toBe(B);expect(other.result.code).not.toBe(first.result.code);
    const state=await snapshot();expect(state.codes.find(r=>r.code===first.result.code)!.issued_to).toBe(A);expect(state.codes.find(r=>r.code===other.result.code)!.issued_to).toBe(B);
  });
});
