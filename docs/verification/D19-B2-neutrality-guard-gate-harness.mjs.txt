import {readFileSync,writeFileSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import process from 'node:process';
import os from 'node:os';
const hash=v=>createHash('sha256').update(v).digest('hex');
const binding=JSON.parse(readFileSync('docs/verification/code-fingerprint.json','utf8'));
const source=()=>hash(JSON.stringify(binding.files.map(f=>({path:f.path,sha256:hash(readFileSync(f.path,'utf8').replaceAll('\r\n','\n'))}))));
if(source()!==binding.fingerprint)throw Error('Current source differs from sealed candidate');
const env={...process.env};
for(const key of Object.keys(env))if(/SUPABASE|INTEGRATION|ADFOCATE|OPENAI|OPENROUTER|ANTHROPIC|GEMINI|RESEND|TWILIO|WHATSAPP|SMTP|SENDGRID/.test(key))env[key]='';
Object.assign(env,{RUN_INTEGRATION:'0',RUN_AUDIT_INTEGRATION:'0',NODE_OPTIONS:''});
const attempt=process.argv[3]??'1';
const selected=process.argv[2],commands={type:['run','type-check'],lint:['run','lint'],manifest:['run','release:manifest:check'],full:['test','--','--no-file-parallelism','--reporter=json','--outputFile=.codex-tmp/B2-neutrality-guard-full-attempt1.json'],build:['run','build']};
for(const key of Object.keys(commands))commands[key]=commands[key].map(v=>v.replaceAll('attempt1','attempt'+attempt));
if(!commands[selected])throw Error('Unknown offline gate');
const startedAt=new Date().toISOString(),freeGiBBefore=os.freemem()/2**30;
const npmPath=process.execPath.replace(/node\.exe$/i,'node_modules/npm/bin/npm-cli.js');
const result=spawnSync(process.execPath,[npmPath,...commands[selected]],{env,encoding:'utf8',timeout:1200000,maxBuffer:64*1024*1024});
const output=(result.stdout??'')+(result.stderr??'');writeFileSync(`.codex-tmp/B2-neutrality-guard-${selected}-attempt${attempt}.log`,output);
const receipt={gate:selected,command:`npm ${commands[selected].join(' ')}`,sourceFingerprint:binding.fingerprint,sourceRecords:binding.files.length,sourceUnchangedAfter:source()===binding.fingerprint,startedAt,finishedAt:new Date().toISOString(),exit:result.status,error:result.error?.code??null,logSha256:hash(output),freeGiBBefore,freeGiBAfter:os.freemem()/2**30,offline:true,realTargetIntegrationEnabled:false,providerKeysSupplied:false};
writeFileSync(`.codex-tmp/B2-neutrality-guard-${selected}-attempt${attempt}-receipt.json`,JSON.stringify(receipt,null,2)+'\n');process.stdout.write(JSON.stringify(receipt)+'\n');if(result.status!==0)process.stderr.write(output.slice(-14000));process.exitCode=result.status??1;
