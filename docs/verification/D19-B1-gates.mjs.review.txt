import {readFileSync,writeFileSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import process from 'node:process';
import os from 'node:os';
const hash=v=>createHash('sha256').update(v).digest('hex');
const old=JSON.parse(readFileSync('docs/verification/code-fingerprint.json','utf8'));
const paths=[...new Set([...old.files.map(f=>f.path),'server/test/acceptance-target-retirement.test.ts'])].sort();
const source=()=>{const files=paths.map(path=>({path,sha256:hash(readFileSync(path,'utf8').replaceAll('\r\n','\n'))}));return {method:old.method+'; B1 extends the existing acceptance identity guard and its target helper; adds the offline retirement regression.',headSha:old.headSha,fingerprint:hash(JSON.stringify(files)),files};};
const binding=source();writeFileSync('.codex-tmp/D19-B1-source-after.json',JSON.stringify(binding,null,2)+'\n');
const env={...process.env,RUN_INTEGRATION:'0',SUPABASE_URL:'',SUPABASE_SERVICE_KEY:'',SUPABASE_ANON_KEY:''};
for(const key of Object.keys(env))if(/SUPABASE|INTEGRATION|BROWSER_SCOPE|OPENAI|OPENROUTER|ANTHROPIC|GEMINI|RESEND|TWILIO|WHATSAPP|SMTP|SENDGRID/.test(key))env[key]='';
env.RUN_INTEGRATION='0';
const selected=process.argv[2];
const commands={type:['run','type-check'],lint:['run','lint'],manifest:['run','release:manifest:check'],full:['test','--','--no-file-parallelism','--reporter=json','--outputFile=.codex-tmp/D19-B1-full-verified.json'],build:['run','build']};
commands.environment=['test','--','--no-file-parallelism','server/devApi.test.ts','server/devApi.transport.test.ts','scripts/release/manifest.test.ts','--reporter=json','--outputFile=.codex-tmp/D19-B1-environment.json'];
// Resolve the installed npm CLI without asking a shell to interpret commands.
const npmPath=process.execPath.replace(/node\.exe$/i,'node_modules/npm/bin/npm-cli.js');
if(!commands[selected])throw Error('Unknown local gate');
const startedAt=new Date().toISOString();
const freeGiBBefore=os.freemem()/2**30;
const result=spawnSync(process.execPath,[npmPath,...commands[selected]],{env,encoding:'utf8',timeout:1200000,maxBuffer:32*1024*1024});
const text=(result.stdout??'')+(result.stderr??'');
writeFileSync(`.codex-tmp/D19-B1-${selected}.log`,text);
const receipt={gate:selected,command:`npm ${commands[selected].join(' ')}`,sourceFingerprint:binding.fingerprint,sourceRecords:binding.files.length,sourceUnchangedAfter:source().fingerprint===binding.fingerprint,startedAt,finishedAt:new Date().toISOString(),exit:result.status,error:result.error?.code??null,logSha256:hash(text),freeGiBBefore,freeGiBAfter:os.freemem()/2**30,offline:true,integrationEnabled:false,providerKeysSupplied:false};
writeFileSync(`.codex-tmp/D19-B1-${selected}-receipt.json`,JSON.stringify(receipt,null,2)+'\n');
process.stdout.write(JSON.stringify(receipt)+'\n');
if(result.status!==0)process.stderr.write(text.slice(-9000));
process.exitCode=result.status??1;
